## Goal

Implement `src/surveillance/ingestion/gadgetbridge_parser.py` — reads Gadgetbridge auto-export `.db` SQLite files, extracts all 7 health metric types, transforms to normalized JSON health event dicts.

---

## Context

Gadgetbridge exports SQLite with its own schema. This parser translates to Guinevere's normalized event format. Design: pure function (`.db` in → `list[dict]` out), no side effects, no DB writes. Gadgetbridge schema maps: `HEART_RATE_SAMPLE` → heart_rate, `SLEEP_SAMPLE` → sleep, `ACTIVITY_SAMPLE` → steps + HR fallback, `SPO2_SAMPLE` → spo2, `STRESS_SAMPLE` → stress, `HRV_SAMPLE` → hrv, `BODY_ENERGY_SAMPLE` → body_battery. Graceful degradation: unknown tables/columns → warnings, not errors. Timestamps normalized to UTC ISO 8601.

```python
"""Gadgetbridge SQLite auto-export parser — extracts 7 health metric types."""
from __future__ import annotations
import logging, sqlite3
from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

logger = logging.getLogger(__name__)

OWNER_ID = "00000000-0000-0000-0000-000000000001"
DEFAULT_DEVICE_ID = "watch_s3_01"
SOURCE = "gadgetbridge"

EVENT_TYPE_MAP: dict[str, str] = {}
for _t in ("heart_rate","sleep","steps","spo2","stress","hrv","body_battery"):
    EVENT_TYPE_MAP[_t] = "surveillance.wearable.health.explicit"

STAGE_MAP = {1: "deep", 2: "light", 3: "rem", 4: "awake", 5: "unknown"}
REQUIRED_TABLES = {"DEVICE", "ACTIVITY_SAMPLE"}


@dataclass
class ParseResult:
    events: list[dict[str, Any]] = field(default_factory=list)
    device_id: str = DEFAULT_DEVICE_ID
    device_name: str = ""
    records_parsed: int = 0
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    schema_version: str = "unknown"

    @property
    def success(self) -> bool:
        return len(self.errors) == 0 and self.records_parsed > 0


def _ensure_utc(ts: Any) -> str | None:
    if ts is None:
        return None
    if isinstance(ts, (int, float)):
        dt = datetime.fromtimestamp(ts / 1000 if ts > 1e12 else ts, tz=timezone.utc)
    elif isinstance(ts, str):
        for fmt in ("%Y-%m-%d %H:%M:%S", "%Y-%m-%dT%H:%M:%S"):
            try:
                dt = datetime.strptime(ts, fmt).replace(tzinfo=timezone.utc)
                break
            except ValueError:
                continue
        else:
            return ts
    elif isinstance(ts, datetime):
        dt = ts if ts.tzinfo else ts.replace(tzinfo=timezone.utc)
    else:
        return str(ts)
    return dt.isoformat()


def _evt(event_type: str, ts: str, device_id: str, payload: dict[str, Any]) -> dict[str, Any]:
    return {"event_type": event_type, "scope": EVENT_TYPE_MAP[event_type],
            "timestamp": ts, "device_id": device_id, "source": SOURCE,
            "owner_id": OWNER_ID, "payload": payload}


def _validate_db(db_path: Path, result: ParseResult) -> sqlite3.Connection | None:
    if not db_path.exists():
        result.errors.append(f"File not found: {db_path}"); return None
    try:
        conn = sqlite3.connect(str(db_path))
        conn.row_factory = sqlite3.Row
        ok = conn.execute("PRAGMA integrity_check;").fetchone()
        if ok and ok[0] != "ok":
            result.errors.append(f"Integrity fail: {ok[0]}"); conn.close(); return None
        tables = {r[0] for r in conn.execute("SELECT name FROM sqlite_master WHERE type='table';")}
        missing = REQUIRED_TABLES - tables
        if missing:
            result.errors.append(f"Missing required: {missing}"); conn.close(); return None
        return conn
    except sqlite3.Error as exc:
        result.errors.append(f"DB error: {exc}"); return None


def _extract_device(conn: sqlite3.Connection, result: ParseResult) -> None:
    try:
        row = conn.execute("SELECT NAME, IDENTIFIER FROM DEVICE LIMIT 1;").fetchone()
        if row:
            result.device_name = row["NAME"] or "unknown"
            if row["IDENTIFIER"]:
                result.device_id = row["IDENTIFIER"]
    except sqlite3.Error as exc:
        result.warnings.append(f"DEVICE error: {exc}")


def _extract_hr(conn: sqlite3.Connection, result: ParseResult) -> None:
    for q in ("SELECT TIMESTAMP, HEART_RATE FROM HEART_RATE_SAMPLE WHERE HEART_RATE>0;",
              "SELECT TIMESTAMP, HEART_RATE FROM ACTIVITY_SAMPLE WHERE HEART_RATE>0;"):
        try:
            for row in conn.execute(q).fetchall():
                if ts := _ensure_utc(row["TIMESTAMP"]):
                    result.events.append(_evt("heart_rate", ts, result.device_id, {"bpm": row["HEART_RATE"]}))
                    result.records_parsed += 1
            return
        except sqlite3.Error:
            continue
    result.warnings.append("No heart rate data")


def _extract_sleep(conn: sqlite3.Connection, result: ParseResult) -> None:
    try:
        rows = conn.execute("SELECT TIMESTAMP, SLEEP_STATE, DURATION FROM SLEEP_SAMPLE WHERE SLEEP_STATE IS NOT NULL;").fetchall()
        if not rows:
            result.warnings.append("No sleep data"); return
        for row in rows:
            if ts := _ensure_utc(row["TIMESTAMP"]):
                stage = STAGE_MAP.get(row["SLEEP_STATE"], "unknown")
                result.events.append(_evt("sleep", ts, result.device_id, {"stage": stage, "duration_s": row["DURATION"] or 0}))
                result.records_parsed += 1
    except sqlite3.Error as exc:
        result.warnings.append(f"Sleep error: {exc}")


def _extract_steps(conn: sqlite3.Connection, result: ParseResult) -> None:
    try:
        for row in conn.execute("SELECT TIMESTAMP, STEPS FROM ACTIVITY_SAMPLE WHERE STEPS>0;").fetchall():
            if ts := _ensure_utc(row["TIMESTAMP"]):
                result.events.append(_evt("steps", ts, result.device_id, {"steps": row["STEPS"]}))
                result.records_parsed += 1
    except sqlite3.Error as exc:
        result.warnings.append(f"Steps error: {exc}")


def _extract_spo2(conn: sqlite3.Connection, result: ParseResult) -> None:
    try:
        for row in conn.execute("SELECT TIMESTAMP, SPO2 FROM SPO2_SAMPLE WHERE SPO2>0;").fetchall():
            if ts := _ensure_utc(row["TIMESTAMP"]):
                result.events.append(_evt("spo2", ts, result.device_id, {"spo2_pct": float(row["SPO2"])}))
                result.records_parsed += 1
    except sqlite3.Error:
        result.warnings.append("No SpO2 data")


def _extract_stress(conn: sqlite3.Connection, result: ParseResult) -> None:
    try:
        for row in conn.execute("SELECT TIMESTAMP, STRESS FROM STRESS_SAMPLE WHERE STRESS BETWEEN 0 AND 100;").fetchall():
            if ts := _ensure_utc(row["TIMESTAMP"]):
                result.events.append(_evt("stress", ts, result.device_id, {"stress_score": row["STRESS"]}))
                result.records_parsed += 1
    except sqlite3.Error:
        result.warnings.append("No stress data")


def _extract_hrv(conn: sqlite3.Connection, result: ParseResult) -> None:
    try:
        for row in conn.execute("SELECT TIMESTAMP, VALUE FROM HRV_SAMPLE WHERE VALUE>0;").fetchall():
            if ts := _ensure_utc(row["TIMESTAMP"]):
                result.events.append(_evt("hrv", ts, result.device_id, {"rmssd_ms": float(row["VALUE"])}))
                result.records_parsed += 1
    except sqlite3.Error:
        result.warnings.append("No HRV data")


def _extract_bb(conn: sqlite3.Connection, result: ParseResult) -> None:
    try:
        for row in conn.execute("SELECT TIMESTAMP, ENERGY FROM BODY_ENERGY_SAMPLE WHERE ENERGY BETWEEN 0 AND 100;").fetchall():
            if ts := _ensure_utc(row["TIMESTAMP"]):
                result.events.append(_evt("body_battery", ts, result.device_id, {"energy_score": row["ENERGY"]}))
                result.records_parsed += 1
    except sqlite3.Error:
        result.warnings.append("No body battery data")


def parse_gadgetbridge_db(db_path: str | Path) -> ParseResult:
    """Parse Gadgetbridge .db into health event dicts. Returns ParseResult."""
    db_path = Path(db_path)
    result = ParseResult()
    logger.info("Parsing: %s", db_path)
    conn = _validate_db(db_path, result)
    if conn is None:
        return result
    try:
        _extract_device(conn, result)
        for fn in (_extract_hr, _extract_sleep, _extract_steps, _extract_spo2, _extract_stress, _extract_hrv, _extract_bb):
            fn(conn, result)
        logger.info("Parsed %d events (w:%d e:%d)", result.records_parsed, len(result.warnings), len(result.errors))
    finally:
        conn.close()
    return result


def batch_parse_directory(directory: str | Path, pattern: str = "*.db") -> list[ParseResult]:
    """Parse all .db files in a directory."""
    results: list[ParseResult] = []
    for db_file in sorted(Path(directory).glob(pattern)):
        results.append(parse_gadgetbridge_db(db_file))
    return results
```

"""Tests for gadgetbridge_parser.py"""
import sqlite3
from datetime import datetime, timedelta, timezone
from pathlib import Path
import pytest
from src.surveillance.ingestion.gadgetbridge_parser import parse_gadgetbridge_db, batch_parse_directory


def _make_db(path: Path, full: bool = True) -> None:
    c = sqlite3.connect(str(path))
    c.execute("CREATE TABLE DEVICE (NAME TEXT, IDENTIFIER TEXT);")
    c.execute("INSERT INTO DEVICE VALUES ('Xiaomi Watch S3','watch_s3_01');")
    c.execute("CREATE TABLE ACTIVITY_SAMPLE (TIMESTAMP INTEGER, STEPS INTEGER, HEART_RATE INTEGER);")
    c.execute("CREATE TABLE HEART_RATE_SAMPLE (TIMESTAMP INTEGER, HEART_RATE INTEGER);")
    c.execute("CREATE TABLE SLEEP_SAMPLE (TIMESTAMP INTEGER, SLEEP_STATE INTEGER, DURATION INTEGER);")
    now = datetime.now(timezone.utc)
    for h in range(24):
        ts = int((now - timedelta(hours=h)).timestamp() * 1000)
        c.execute("INSERT INTO HEART_RATE_SAMPLE VALUES (?,?)", (ts, 60 + h % 15))
        c.execute("INSERT INTO ACTIVITY_SAMPLE VALUES (?,?,NULL)", (ts, 50 + h * 10))
    ss = int((now - timedelta(hours=8)).timestamp() * 1000)
    for st, dur in [(1,3600),(2,5400),(3,1800),(4,600)]:
        c.execute("INSERT INTO SLEEP_SAMPLE VALUES (?,?,?)", (ss, st, dur)); ss += dur * 1000
    if full:
        for tbl, cols, val in [
            ("SPO2_SAMPLE","TIMESTAMP INTEGER, SPO2 REAL",(int(now.timestamp()*1000),98.5)),
            ("STRESS_SAMPLE","TIMESTAMP INTEGER, STRESS INTEGER",(int(now.timestamp()*1000),35)),
            ("HRV_SAMPLE","TIMESTAMP INTEGER, VALUE REAL",(int(now.timestamp()*1000),42.5)),
            ("BODY_ENERGY_SAMPLE","TIMESTAMP INTEGER, ENERGY INTEGER",(int(now.timestamp()*1000),78)),
        ]:
            c.execute(f"CREATE TABLE {tbl} ({cols});")
            c.execute(f"INSERT INTO {tbl} VALUES (?,?)", val)
    c.commit(); c.close()


class TestParser:
    def test_valid(self, tmp_path: Path) -> None:
        r = parse_gadgetbridge_db(_path := tmp_path / "ok.db"); _make_db(_path)
        r = parse_gadgetbridge_db(_path)
        assert r.success and r.device_name == "Xiaomi Watch S3"
        assert {e["event_type"] for e in r.events} == {"heart_rate","sleep","steps","spo2","stress","hrv","body_battery"}

    def test_missing(self, tmp_path: Path) -> None:
        r = parse_gadgetbridge_db(tmp_path / "nope.db")
        assert not r.success and "not found" in str(r.errors[0]).lower()

    def test_corrupt(self, tmp_path: Path) -> None:
        (p := tmp_path / "bad.db").write_text("junk")
        assert not parse_gadgetbridge_db(p).success

    def test_partial(self, tmp_path: Path) -> None:
        _make_db(p := tmp_path / "partial.db", full=False)
        r = parse_gadgetbridge_db(p)
        assert r.success and len(r.warnings) >= 3

    def test_payload_keys(self, tmp_path: Path) -> None:
        _make_db(p := tmp_path / "keys.db", full=True)
        want = {"heart_rate":{"bpm"},"sleep":{"stage","duration_s"},"steps":{"steps"},"spo2":{"spo2_pct"},"stress":{"stress_score"},"hrv":{"rmssd_ms"},"body_battery":{"energy_score"}}
        for e in parse_gadgetbridge_db(p).events:
            assert set(e["payload"]) == want[e["event_type"]]

    def test_batch(self, tmp_path: Path) -> None:
        for i in range(3): _make_db(tmp_path / f"b{i}.db")
        assert len(batch_parse_directory(tmp_path)) == 3

    def test_owner_id(self, tmp_path: Path) -> None:
        _make_db(p := tmp_path / "owner.db")
        for e in parse_gadgetbridge_db(p).events:
            assert e["owner_id"] == "00000000-0000-0000-0000-000000000001"

## Troubleshooting

- **`ModuleNotFoundError: No module named 'src'`**: `export PYTHONPATH=/home/guinevere/code/guinevere` or `pip install -e .`
- **`no such table: HEART_RATE_SAMPLE`**: Expected for Gadgetbridge < 0.80 — falls back to `ACTIVITY_SAMPLE` (warning, not error)
- **Timestamps in WIB, not UTC**: Gadgetbridge stores device-local time. `_ensure_utc` treats unix ts as UTC.
- **SPO2 integers (98 vs 98.5)**: Casts to float regardless — valid for `health.spo2` REAL column
- **MemoryError on >1GB .db files**: In practice hourly exports are ~100–500KB. For large backfills, implement streaming iterator pattern.
