**Estimated Time:** 3 hours

#### Context

Current health check from P0/P8 is basic — returns 200 OK with no dependency validation. Production hardening requires deep health checks that validate every downstream dependency: PostgreSQL (SELECT 1 via PgBouncer connection pool with 3s timeout), Redis (PING with 2s timeout), LLM API (9Router health endpoint with 5s timeout), disk space (>20% free on /var and /data volumes), and memory usage (below cgroup limit from systemd MemoryMax). Exposes Prometheus metrics at /metrics for Grafana scraping. Deep health check feeds P10-007 (CI/CD deploy gate) and P10-010 (rollback health check). Shallow health check used by systemd ExecStartPost and load balancers. AC-CORE-006 enforces runtime config validation — health check must fail closed if required config keys are missing.

@dataclass
class CheckResult:
    name: str
    status: str  # "healthy" | "degraded" | "unhealthy"
    latency_ms: float
    detail: str = ""
    error: Optional[str] = None


async def _check_postgresql(timeout: float = 3.0) -> CheckResult:
    """Validate PostgreSQL connectivity via PgBouncer pool with SELECT 1."""
    start = time.monotonic()
    try:
        pool = await get_pg_pool()
        async with pool.acquire() as conn:
            async with conn.cursor() as cur:
                await asyncio.wait_for(
                    cur.execute("SELECT 1"),
                    timeout=timeout,
                )
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="postgresql",
            status="healthy",
            latency_ms=round(latency, 2),
            detail="SELECT 1 via PgBouncer pool",
        )
    except asyncio.TimeoutError:
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="postgresql",
            status="unhealthy",
            latency_ms=round(latency, 2),
            error=f"Timeout after {timeout}s",
        )
    except Exception as exc:
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="postgresql",
            status="unhealthy",
            latency_ms=round(latency, 2),
            error=str(exc),
        )


async def _check_redis(timeout: float = 2.0) -> CheckResult:
    """Validate Redis connectivity and capture metrics."""
    start = time.monotonic()
    settings = get_settings()
    try:
        redis_client = Redis(
            host="127.0.0.1",
            port=6380,
            username="guinevere_core",
            password=settings.redis_core_password,
            socket_timeout=timeout,
        )
        pong = await asyncio.wait_for(redis_client.ping(), timeout=timeout)
        assert pong is True, "Redis PING returned non-True"

        info = await redis_client.info("memory")
        redis_memory_used.set(float(info.get("used_memory", 0)))

        clients_info = await redis_client.info("clients")
        redis_connected_clients.set(
            float(clients_info.get("connected_clients", 0))
        )

        await redis_client.close()
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="redis",
            status="healthy",
            latency_ms=round(latency, 2),
            detail=f"PONG, memory={info.get('used_memory_human', 'N/A')}",
        )
    except asyncio.TimeoutError:
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="redis",
            status="unhealthy",
            latency_ms=round(latency, 2),
            error=f"Timeout after {timeout}s",
        )
    except Exception as exc:
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="redis",
            status="unhealthy",
            latency_ms=round(latency, 2),
            error=str(exc),
        )


async def _check_llm_api(timeout: float = 5.0) -> CheckResult:
    """Validate LLM 9Router reachability via health endpoint.

    LLM API check is non-blocking per design: if 9Router is unreachable,
    overall status becomes 'degraded' not 'unhealthy'.
    """
    start = time.monotonic()
    settings = get_settings()
    try:
        import httpx
        llm_health_url = settings.llm_health_url
        async with httpx.AsyncClient(timeout=timeout) as client:
            resp = await client.get(llm_health_url)
        if resp.status_code == 200:
            llm_api_reachable.set(1)
            latency = (time.monotonic() - start) * 1000
            return CheckResult(
                name="llm_api",
                status="healthy",
                latency_ms=round(latency, 2),
                detail=f"9Router health: {resp.status_code}",
            )
        else:
            llm_api_reachable.set(0)
            latency = (time.monotonic() - start) * 1000
            return CheckResult(
                name="llm_api",
                status="degraded",
                latency_ms=round(latency, 2),
                error=f"9Router returned {resp.status_code}",
            )
    except asyncio.TimeoutError:
        llm_api_reachable.set(0)
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="llm_api",
            status="degraded",
            latency_ms=round(latency, 2),
            error=f"Timeout after {timeout}s",
        )
    except Exception as exc:
        llm_api_reachable.set(0)
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="llm_api",
            status="degraded",
            latency_ms=round(latency, 2),
            error=str(exc),
        )


def _check_disk(required_free_pct: float = 20.0) -> CheckResult:
    """Check disk space on critical mount points."""
    start = time.monotonic()
    mounts = ["/var", "/data"]
    errors: List[str] = []
    all_healthy = True

    for mount in mounts:
        try:
            usage = psutil.disk_usage(mount)
            free = usage.free / usage.total * 100
            disk_free_pct.labels(mount=mount).set(free)
            if free < required_free_pct:
                all_healthy = False
                errors.append(
                    f"{mount}: {free:.1f}% free (threshold: {required_free_pct}%)"
                )
        except FileNotFoundError:
            errors.append(f"{mount}: mount point not found")
            all_healthy = False

    latency = (time.monotonic() - start) * 1000
    if all_healthy:
        return CheckResult(
            name="disk",
            status="healthy",
            latency_ms=round(latency, 2),
            detail=f"All mounts above {required_free_pct}% free",
        )
    else:
        return CheckResult(
            name="disk",
            status="unhealthy" if len(errors) >= 2 else "degraded",
            latency_ms=round(latency, 2),
            error="; ".join(errors),
        )


def _check_memory() -> CheckResult:
    """Check process memory against system limits."""
    start = time.monotonic()
    try:
        process = psutil.Process(os.getpid())
        mem_info = process.memory_info()
        mem_bytes = mem_info.rss

        cgroup_limit: Optional[int] = None
        cgroup_path = "/sys/fs/cgroup/memory.current"
        limit_path = "/sys/fs/cgroup/memory.max"
        if os.path.exists(cgroup_path) and os.path.exists(limit_path):
            with open(limit_path, "r") as f:
                limit_str = f.read().strip()
                if limit_str != "max":
                    cgroup_limit = int(limit_str)

        if cgroup_limit:
            pct = mem_bytes / cgroup_limit * 100
            memory_used_pct.set(pct)
            detail = f"{pct:.1f}% of cgroup limit ({mem_bytes >> 20}MB / {cgroup_limit >> 20}MB)"
        else:
            memory_used_pct.set(0)
            detail = f"RSS: {mem_bytes >> 20}MB (no cgroup limit set)"

        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="memory",
            status="healthy",
            latency_ms=round(latency, 2),
            detail=detail,
        )
    except Exception as exc:
        latency = (time.monotonic() - start) * 1000
        return CheckResult(
            name="memory",
            status="degraded",
            latency_ms=round(latency, 2),
            error=str(exc),
        )


async def _check_pool_metrics() -> None:
    """Update pool metric gauges from PgBouncer admin console.

    Called during deep health check to refresh Prometheus gauges.
    Requires PgBouncer admin connection on port 6432.
    """
    try:
        import asyncpg
        conn = await asyncio.wait_for(
            asyncpg.connect(
                host="127.0.0.1",
                port=6432,
                user="pgbouncer",
                database="pgbouncer",
            ),
            timeout=3.0,
        )
        try:
            pools = await conn.fetch("SHOW POOLS")
            total_active = sum(
                int(row["cl_active"]) for row in pools if row["database"] != "pgbouncer"
            )
            total_waiting = sum(
                int(row["cl_waiting"]) for row in pools if row["database"] != "pgbouncer"
            )
            pg_pool_active.set(total_active)
            pg_pool_waiting.set(total_waiting)
        finally:
            await conn.close()
    except Exception:
        # Pool metrics are best-effort; failures don't block health check
        pass

async def health_shallow() -> Dict[str, str]:

    Used by systemd ExecStartPost and load balancers.
    Must respond in <100ms.
    """
    return {"status": "alive"}


@router.get("/health/deep")
async def health_deep() -> Dict[str, Any]:
    """Deep readiness probe — validates all downstream dependencies.

    Checks PostgreSQL, Redis, LLM API, disk space, memory.
    Returns aggregated status: healthy, degraded, or unhealthy.
    """
    start = time.monotonic()

    # Run all checks concurrently (each with independent timeout)
    results: List[CheckResult] = await asyncio.gather(
        _check_postgresql(),
        _check_redis(),
        _check_llm_api(),
        return_exceptions=False,
    )

    # Disk and memory are synchronous (fast, local checks)
    results.append(_check_disk())
    results.append(_check_memory())

    # Update pool metrics gauges (best-effort)
    await _check_pool_metrics()

    # Determine aggregate status
    statuses = [r.status for r in results]
    if "unhealthy" in statuses:
        aggregate = "unhealthy"
        health_status_gauge.set(0)
    elif "degraded" in statuses:
        aggregate = "degraded"
        health_status_gauge.set(1)
    else:
        aggregate = "healthy"
        health_status_gauge.set(2)

    # Update per-check duration gauges
    for r in results:
        health_check_duration.labels(check_name=r.name).set(r.latency_ms / 1000)

    total_latency = (time.monotonic() - start) * 1000

    return {
        "status": aggregate,
        "timestamp": time.time(),
        "total_latency_ms": round(total_latency, 2),
        "checks": {
            r.name: {
                "status": r.status,
                "latency_ms": r.latency_ms,
                "detail": r.detail,
                "error": r.error,
            }
            for r in results
        },
    }

# --- Systemd Watchdog Integration ---
def notify_watchdog() -> None:
    """Send WATCHDOG=1 notification to systemd.

    Called periodically (e.g., every WATCHDOG_USEC/2) from the main loop.
    systemd kills the process if notifications stop for WATCHDOG_USEC.
    """
    watchdog_pid = os.environ.get("WATCHDOG_PID")
    if watchdog_pid and int(watchdog_pid) == os.getpid():
        import socket
        sock = socket.socket(socket.AF_UNIX, socket.SOCK_DGRAM)
        try:
            sock.sendto(b"WATCHDOG=1", os.environ.get("NOTIFY_SOCKET", ""))
        except Exception:
            pass
        finally:
            sock.close()


async def periodic_watchdog(interval: float = 30.0) -> None:
    """Background task: send watchdog notifications every `interval` seconds."""
    while True:
        notify_watchdog()
        await asyncio.sleep(interval)

#### Troubleshooting
- **Issue:** GET /health/deep hangs forever on PostgreSQL check
  - **Solution:** Verify timeout is properly applied to the asyncpg query. Ensure PgBouncer is running: `systemctl is-active pgbouncer`. If PgBouncer pool is exhausted, increase pool size in pgbouncer.ini
- **Issue:** Prometheus metrics endpoint returns empty or 404
  - **Solution:** Verify `prometheus-client` package installed: `uv run python -c "import prometheus_client"`. Check router registration in app.py. Ensure metrics route is registered before any middleware that might interfere
- **Issue:** LLM API check always returns "degraded" even when 9Router is up
  - **Solution:** Verify LLM health URL in config: `print(get_settings().llm_health_url)`. Test manually: `curl -v <llm_health_url>`. Check Tailscale connectivity: `tailscale status`. Ensure 9Router is running on the target host
- **Issue:** Disk check returns "unhealthy" but df shows plenty of space
  - **Solution:** The check only monitors /var and /data mounts. If one doesn't exist, the check fails. Verify mount points: `df -h /var /data`. If /data doesn't exist, remove it from the mounts list in `_check_disk()`
- **Issue:** `disk_free_pct` label cardinality explosion
  - **Solution:** The label "mount" has only 2 values (/var and /data) — this is fixed cardinality, not a problem. If you add dynamic mount points, ensure they're limited to a small set
- **Issue:** Systemd watchdog kills process even though health checks pass
  - **Solution:** Verify `WATCHDOG_USEC` and `WatchdogSec` are > 2x the `periodic_watchdog` interval (30s watchdog interval, 60s WatchdogSec is safe). Check journal for WATCHDOG notifications: `journalctl -u guinevere-gateway -f | grep -i watchdog`

#### Notes
- Deep health check used by P10-007 (CI/CD deploy gate — verifies health after deploy) and P10-010 (rollback — verifies health after rollback)
- Shallow health check used by systemd ExecStartPost, Caddy reverse proxy health check, and Kubernetes-style liveness probes
- LLM API check is intentionally non-blocking: 9Router unreachability sets status to "degraded" not "unhealthy" because Guinevere can still serve cached/offline responses
- Disk check monitors /var (logs, application) and /data (PostgreSQL data directory). If your deployment uses different mount points, update the mounts list
- Pool metrics (pg_pool_active, pg_pool_waiting) require PgBouncer admin access on port 6432. If PgBouncer is not used, remove these gauges and the `_check_pool_metrics()` function
- The Prometheus `METRICS_REGISTRY` is isolated (not using `REGISTRY` global) to avoid duplicate registration errors when modules are reloaded
- Watchdog integration requires systemd version >= 236 and `WatchdogSec=` set in the service unit. The `periodic_watchdog()` task sends notifications at half the WatchdogSec interval for safety margin
