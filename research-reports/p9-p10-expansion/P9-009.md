#### Context

The `/finance summary` command (P9-008) provides a quick overview. This step adds a detailed `/finance report` subcommand for deep analysis. The command queries `financial.transactions` with GROUP BY aggregations across category, merchant, and date dimensions. Output is rendered as a multi-field Discord embed with formatted tables, plus an optional CSV attachment generated via Python `tempfile` + `discord.File`. The report supports three time range modes: `current_month` (default), `last_month`, and a custom `date_range` (start/end date string parameters). All computations happen server-side; the bot sends a defer response immediately for queries expected to take >2 seconds. Permission is restricted to the owner (Faiz) via the existing bot permission decorator.

"""Discord slash command: /finance report — detailed financial breakdown."""

import csv
import io
import tempfile
from datetime import datetime, timedelta
from typing import Optional, Literal

import asyncpg
import discord
from discord import app_commands
from dateutil.relativedelta import relativedelta

from src.bot.utils.permissions import owner_only
from src.financial.queries import (
    get_category_breakdown,
    get_top_merchants,
    get_daily_spending_trend,
    get_month_over_month_comparison,
    get_budget_utilization,
)

PeriodType = Literal["current_month", "last_month", "custom"]


class FinanceReportCommand(app_commands.Group):
    """Detailed financial report subcommand group."""

    def __init__(self, pool: asyncpg.Pool):
        super().__init__(name="finance", description="Financial tracking commands")
        self.pool = pool

    @app_commands.command(name="report", description="Generate detailed financial report")
    @app_commands.describe(
        period="Time range for the report",
        start_date="Start date (YYYY-MM-DD) — required for custom period",
        end_date="End date (YYYY-MM-DD) — required for custom period",
        export_csv="Attach a CSV file with the full transaction list",
    )
    @owner_only()
    async def finance_report(
        self,
        interaction: discord.Interaction,
        period: Literal["current_month", "last_month", "custom"] = "current_month",
        start_date: Optional[str] = None,
        end_date: Optional[str] = None,
        export_csv: bool = False,
    ) -> None:
        await interaction.response.defer(thinking=True)

        date_range = self._resolve_date_range(period, start_date, end_date)
        if date_range is None:
            await interaction.followup.send(
                "Error: `start_date` and `end_date` required for custom period (format: YYYY-MM-DD)."
            )
            return

        start, end = date_range

        async with self.pool.acquire() as conn:
            cat_breakdown = await get_category_breakdown(conn, start, end)
            top_merchants = await get_top_merchants(conn, start, end, limit=10)
            daily_trend = await get_daily_spending_trend(conn, start, end)
            mom_comparison = await get_month_over_month_comparison(conn, start, end)
            budget_util = await get_budget_utilization(conn, start, end)

        embed = self._build_embed(start, end, cat_breakdown, top_merchants, daily_trend, mom_comparison, budget_util)

        files = []
        if export_csv:
            csv_path = self._generate_csv(start, end, cat_breakdown, top_merchants, daily_trend)
            files.append(discord.File(csv_path, filename=f"finance_report_{start.date()}_{end.date()}.csv"))

        await interaction.followup.send(embed=embed, files=files if files else None)

        if export_csv and csv_path:
            import os
            os.unlink(csv_path)

    def _resolve_date_range(
        self,
        period: str,
        start_date: Optional[str],
        end_date: Optional[str],
    ) -> Optional[tuple[datetime, datetime]]:
        now = datetime.utcnow()
        if period == "current_month":
            start = now.replace(day=1, hour=0, minute=0, second=0, microsecond=0)
            end = now
        elif period == "last_month":
            first_of_this = now.replace(day=1)
            start = first_of_this - relativedelta(months=1)
            end = first_of_this - timedelta(seconds=1)
        elif period == "custom":
            if not start_date or not end_date:
                return None
            start = datetime.strptime(start_date, "%Y-%m-%d")
            end = datetime.strptime(end_date, "%Y-%m-%d").replace(hour=23, minute=59, second=59)
        else:
            return None
        return (start, end)

    def _build_embed(
        self,
        start: datetime,
        end: datetime,
        cat_breakdown: list[dict],
        top_merchants: list[dict],
        daily_trend: list[dict],
        mom_comparison: dict,
        budget_util: list[dict],
    ) -> discord.Embed:
        total_spend = sum(row["total_amount"] for row in cat_breakdown)
        total_spend_fmt = f"${total_spend:,.2f}"
        date_label = f"{start.strftime('%b %d')} – {end.strftime('%b %d, %Y')}"

        embed = discord.Embed(
            title=f"Financial Report — {date_label}",
            description=f"**Total Spend:** {total_spend_fmt}\n\n*Detailed breakdown below.*",
            color=0x00D2FF,
        )

        if cat_breakdown:
            cat_lines = []
            for row in cat_breakdown:
                pct = (row["total_amount"] / total_spend * 100) if total_spend > 0 else 0
                cat_lines.append(
                    f"`{row['category_name']:<20}` ${row['total_amount']:>8,.2f} ({pct:>5.1f}%)"
                )
            embed.add_field(name="Category Breakdown", value="\n".join(cat_lines[:15]), inline=False)

        if top_merchants:
            merch_lines = []
            for i, row in enumerate(top_merchants, 1):
                merch_lines.append(
                    f"{i}. `{row['merchant_name'][:25]}` — ${row['total_amount']:,.2f} ({row['txn_count']} txns)"
                )
            embed.add_field(name="Top 10 Merchants", value="\n".join(merch_lines), inline=False)

        if daily_trend:
            avg_daily = sum(row["daily_total"] for row in daily_trend) / max(len(daily_trend), 1)
            max_day = max(daily_trend, key=lambda r: r["daily_total"])
            embed.add_field(
                name="Spending Trend",
                value=(
                    f"Avg daily: ${avg_daily:,.2f}\n"
                    f"Highest day: {max_day['date']} (${max_day['daily_total']:,.2f})\n"
                    f"Days tracked: {len(daily_trend)}"
                ),
                inline=False,
            )

        if mom_comparison:
            current = mom_comparison.get("current_total", 0)
            previous = mom_comparison.get("previous_total", 0)
            change_pct = ((current - previous) / previous * 100) if previous > 0 else 0
            direction = "increase" if change_pct > 0 else "decrease"
            embed.add_field(
                name="Month-over-Month",
                value=f"${current:,.2f} vs ${previous:,.2f} ({direction} of {abs(change_pct):.1f}%)",
                inline=False,
            )

        if budget_util:
            budget_lines = []
            for row in budget_util:
                util_pct = row["utilization_pct"]
                flag = "OVER" if util_pct >= 100 else ("WARN" if util_pct >= 80 else "OK")
                budget_lines.append(
                    f"`{row['category_name']:<20}` {util_pct:>5.0f}% of ${row['budget_amount']:,.0f} [{flag}]"
                )
            embed.add_field(name="Budget Utilization", value="\n".join(budget_lines[:10]), inline=False)

        embed.set_footer(text="Generated by Guinevere FinOps | Use /finance report export_csv:True for CSV")
        return embed

    def _generate_csv(
        self,
        start: datetime,
        end: datetime,
        cat_breakdown: list[dict],
        top_merchants: list[dict],
        daily_trend: list[dict],
    ) -> str:
        fd, path = tempfile.mkstemp(suffix=".csv", prefix="finance_report_")
        with os.fdopen(fd, "w", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)

            writer.writerow(["=== CATEGORY BREAKDOWN ==="])
            writer.writerow(["Category", "Total Amount", "Transaction Count", "Avg Amount"])
            for row in cat_breakdown:
                writer.writerow([
                    row["category_name"],
                    f"${row['total_amount']:,.2f}",
                    row["txn_count"],
                    f"${row.get('avg_amount', 0):,.2f}",
                ])

            writer.writerow([])
            writer.writerow(["=== TOP MERCHANTS ==="])
            writer.writerow(["Rank", "Merchant", "Total Amount", "Transaction Count"])
            for i, row in enumerate(top_merchants, 1):
                writer.writerow([i, row["merchant_name"], f"${row['total_amount']:,.2f}", row["txn_count"]])

            writer.writerow([])
            writer.writerow(["=== DAILY TREND ==="])
            writer.writerow(["Date", "Daily Total"])
            for row in daily_trend:
                writer.writerow([str(row["date"]), f"${row['daily_total']:,.2f}"])

        return path

#### Troubleshooting

- **Issue:** Command times out (>3s defer window)
  - **Solution:** Increase defer timeout via `interaction.response.defer(thinking=True, ephemeral=False)`; split heavy queries to run after defer

- **Issue:** CSV file not attached to Discord message
  - **Solution:** Check file size limit (Discord bot limit: 25 MB); verify `tempfile.mkstemp` creates a valid file; ensure file is passed before `followup.send(files=...)`

- **Issue:** Category breakdown shows `None` or missing categories
  - **Solution:** Verify `financial.categories` table has entries for all classification labels; run `SELECT DISTINCT category_name FROM financial.transactions;` to audit

- **Issue:** Month-over-month shows inflated percentages
  - **Solution:** Guard division by zero in `_build_embed`; if `previous_total == 0`, display "N/A (no prior data)" instead of percentage

- **Issue:** Embed exceeds 6000-character Discord limit (fields truncated)
  - **Solution:** Paginate large category breakdowns across multiple fields or split into multiple embeds; limit categories to top 15

#### Notes

- Query functions (`get_category_breakdown`, `get_top_merchants`, etc.) must be implemented in `src/financial/queries.py` as P9-008 pre-work or inline here. Each function accepts `asyncpg.Connection`, `start`, `end` and returns `list[dict]`.
- CSV export uses `tempfile.mkstemp` for atomic temp file creation; file is cleaned up after Discord send via `os.unlink`.
- The `owner_only` decorator should check `interaction.user.id == Faiz's Discord user ID` from config.
- Budget utilization percentages are computed as `(total_spent / budget_amount) * 100` using the `financial.budgets` table.
- Consider caching category lookup results in Redis (TTL: 5 min) if query latency exceeds 500ms with large transaction volumes.
- All monetary values should be stored as PostgreSQL `NUMERIC(12,2)` — do not use float for currency math.
