#!/bin/bash
# 24-hour 9Router soak test
# Logs: success/failure count, latency per request
DURATION=$((24 * 3600))  # 24 hours in seconds
INTERVAL=30  # Check every 30 seconds
SUCCESS=0
FAILURE=0
START=$(date +%s)
LOG="docs/setup-evidence/P10/STEP-P10-021/soak-test-9router.log"

echo "timestamp,status,latency_ms" > "$LOG"

while [ $(($(date +%s) - START)) -lt $DURATION ]; do
    TS=$(date -Iseconds)
    START_MS=$(date +%s%3N)
    HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" \
      --max-time 10 http://100.x.x.x:NNNN/health 2>/dev/null)
    END_MS=$(date +%s%3N)
    LATENCY=$((END_MS - START_MS))

    if [ "$HTTP_CODE" = "200" ]; then
        SUCCESS=$((SUCCESS + 1))
        STATUS="success"
    else
        FAILURE=$((FAILURE + 1))
        STATUS="failure"
    fi

    echo "$TS,$STATUS,$LATENCY" >> "$LOG"

    # Progress every hour
    TOTAL=$((SUCCESS + FAILURE))
    if [ $((TOTAL % 120)) -eq 0 ]; then
        echo "[$(date +%H:%M)] Soak progress: $SUCCESS success, $FAILURE failure (${TOTAL} checks)"
    fi

    sleep $INTERVAL
done

echo "Total: $SUCCESS success, $FAILURE failure"
FAILURE_RATE=$(echo "scale=4; $FAILURE / ($SUCCESS + $FAILURE) * 100" | bc)
echo "Failure rate: ${FAILURE_RATE}%"

if [ "$(echo "$FAILURE_RATE < 1" | bc)" = "1" ]; then
    echo "PASS: Soak test failure rate < 1%"
else
    echo "FAIL: Soak test failure rate >= 1%"
fi
SOAKEOF

#### Troubleshooting
- **Issue:** Flaky tests -- some AC tests pass on one run, fail on another
  - **Solution:** Fix root cause, do NOT skip. Identify the flaky test: run `pytest tests/acceptance/ -v --tb=short --count=5` to repeat and catch intermittent failures. Common causes: race conditions in async tests, dependency on external service state (Redis keys, PostgreSQL data), timing-sensitive assertions
- **Issue:** Cost slightly over $30 at gate check time
  - **Solution:** Investigate and optimize before proceeding. Run `redis-cli -n 5 HGETALL "cost:model_usage:$(date +%Y-%m)"` to identify expensive models. Switch to cheaper model, reduce context window, or implement stricter budget enforcement. Cost must be under $30 to pass FIN criteria
- **Issue:** Persona drift detected (drift score > 0.15 on any day)
  - **Solution:** Tune system prompt in `docs/60-persona/61-SystemPromptMaster_v1.1.md`. Identify what changed: tone? memory recall? safety boundary? Re-deploy with adjusted prompt. Reset the 7-day observation counter and re-observe. Drift must be within bounds for 7 consecutive days
- **Issue:** 9Router failure during soak test (> 1% failure rate)
  - **Solution:** Analyze failure pattern: all failures at specific time? (Background process, cron job causing load). Specific error code? (502 = 9Router crash, 503 = overload, timeout = network). Fix the root cause, reset soak counter, and re-run full 24 hours. Document each failure and fix in soak test log
- **Issue:** Operator sign-off delayed (Faiz unavailable)
  - **Solution:** The gate CANNOT pass without Faiz approval (AC-PHASE-007). While waiting: complete all automatable components (soak test, AC test suite, evidence compilation). Document status as "AWAITING OPERATOR SIGN-OFF". Do NOT begin Phase 11 work
- **Issue:** Aizanta services impacted during soak test on shared VPS
  - **Solution:** Immediately halt soak test. Coordinate with Aizanta operator to schedule during low-usage window or with more conservative resource limits. Document the impact in the report. If resource contention is persistent, consider VPS migration or resource isolation (systemd cgroups, nice levels)
